SELECT
  CAST(P.CreationDate AS date) AS [Date],
  CASE
    WHEN P.DeletionDate IS NULL THEN 'non-deleted'
    ELSE 'deleted'
  END AS Status,
  COUNT(*) AS NewAnswers
FROM PostsWithDeleted P
WHERE P.PostTypeId = 2 -- Answers
  AND P.CreationDate >= '2008-01-01'
GROUP BY
  CAST(P.CreationDate AS date),
  CASE
    WHEN P.DeletionDate IS NULL THEN 'non-deleted'
    ELSE 'deleted'
  END
ORDER BY
  [Date],
  Status;
